## 💾 CÓMO USAR EL CONTENIDO

### En Cualquier Componente Cliente:

```tsx
'use client';

import { useScryvedTranslations } from '@/lib/seo-helpers';

export default function MyComponent() {
  const { services, security, contact } = useScryvedTranslations();
  
  return (
    <div>
      {/* Acceso a cualquier contenido */}
      <h2>{services.title_part1} {services.title_part2}</h2>
      <p>{services.description}</p>
      
      {/* Iterar sobre arrays */}
      {services.items.map(service => (
        <div key={service.id}>
          <h3>{service.title}</h3>
          <p>{service.description}</p>
        </div>
      ))}
    </div>
  );
}
```
